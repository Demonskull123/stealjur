LPH_PRECHECK(function()
    local env = getfenv()
    local a, b = 0x6A09E667, 0xBB67AE85

    local function absorb(value)
        a = bit32.bxor(bit32.lrotate(a, 5), value)
        b = bit32.bxor(bit32.lrotate(b, 11), a)
    end

    for i, v in ipairs({
        LPH_ENCSTR("__PF"), LPH_ENCSTR("__PA"),
        LPH_ENCSTR("__PK"), LPH_ENCSTR("__PID"),
        LPH_ENCSTR("__ENT"), LPH_ENCSTR("__PLAST"),
        LPH_ENCSTR("__SKIPP"), LPH_ENCSTR("__SPIN"),
    }) do
        absorb(env[v] == nil and i * 97 or 0)
    end

    local hooked = env.isfunctionhooked

    for i, v in ipairs({
        debug.info, pcall, xpcall, rawget, rawequal,
        type, string.reverse, string.byte, bit32.bxor,
    }) do
        local ok, source = pcall(debug.info, v, "s")
        local clean = ok and source == "[C]"

        if type(hooked) == "function" then
            local checked, changed = pcall(hooked, v)
            clean = clean and checked and changed == false
        end

        absorb(clean and i * 131 or 0)
    end

    absorb(string.byte(string.reverse("aB09"), 1))
    absorb(bit32.bxor(0x55AA, 0xAA55))

    local ok, result = pcall(function()
        error(LPH_ENCSTR("guard-probe"), 0)
    end)

    absorb(not ok and result == "guard-probe" and 43 or 0)

    return { a, b }
end, { 0x210C79CB, 0x16A721A4 })

        local Index, Amount
        for j, n in pairs(Constants) do
            if type(n) == "number" then
                if Index then Index = nil break end
                Index, Amount = j, n
            end
        end

local Acceleration = function(Value)
    local Scripts = game:GetService("StarterPlayer"):FindFirstChild("StarterPlayerScripts")
    -- The live ClientEngine is under the player. Actor contexts cannot always read
    -- LocalPlayer.PlayerScripts, so guard it and also use the engine above RootScript.
    local Ok, Live = pcall(function()
        return game:GetService("Players").LocalPlayer.PlayerScripts:FindFirstChild("ClientEngine")
    end)
    for i, v in pairs({ Scripts and Scripts:FindFirstChild("ClientEngine"),
        Ok and Live or nil, RootScript:FindFirstAncestor("ClientEngine") }) do
        if typeof(v) == "Instance" then v:SetAttribute("moveAcceleration", Value) end
    end
end
local MobileMovement = function(Value)
    if Data.MobileMove then Data.MobileMove:Disconnect() Data.MobileMove = nil end
    Acceleration(Value and 10000000 or 10)

        Data.MobileMove = game:GetService("RunService").RenderStepped:Connect(function()
            Acceleration(10000000)
        end)
    end
end

    local LucideSet

    local function NormalizeIcon(n)
        return string.lower((string.gsub(n, "[%-_%s]", "")))
    end

    local function GetLucide()
        if LucideSet ~= nil then return LucideSet or nil end
        LucideSet = false
        local Ids = LucideIds
        if type(Ids) == "table" then
            local ByNameNormalized, n = {}, 0
            for Name, Id in pairs(Ids) do
                ByNameNormalized[NormalizeIcon(Name)] = Id
                n += 1
            end
            LucideSet = { byName = Ids, byNorm = ByNameNormalized, count = n }
        end
        return LucideSet or nil
    end

    Interface.CustomIcons = {}
    local RemoteIcons

    local function FetchIcons()
        if RemoteIcons ~= nil then return RemoteIcons end

        local Ok, Result = pcall(function()
            return loadstring(game:HttpGet("https://raw.githubusercontent.com/Footagesus/Icons/refs/heads/main/lucide/dist/Icons.lua"))()
        end)

        RemoteIcons = Ok and type(Result) == "table" and Result or false
        return RemoteIcons
    end

    function Interface.Icon(Name)
        if type(Name) ~= "string" or Name == "" then return nil end
        if string.match(Name, "^rbxasset") or string.match(Name, "^https?://") then return Name end

        local ForceLucide = string.sub(Name, 1, 7) == "lucide:"
        if ForceLucide then Name = string.sub(Name, 8) end
        if Interface.CustomIcons[Name] then return Interface.CustomIcons[Name] end

        local Host = Interface.Env
        if Host and Host.isfile and Host.getcustomasset then
            local Path = Interface.IconFolder .. "/" .. Name .. ".png"
            local Ok, Exists = pcall(Host.isfile, Path)
            if Ok and Exists then
                local Loaded, Image = pcall(Host.getcustomasset, Path)
                if Loaded and type(Image) == "string" then
                    Interface.CustomIcons[Name] = Image
                    return Image
                end
            end
        end

        local Remote = FetchIcons()
        local Icon = Remote and (Remote[Name] or (Remote.Icons and Remote.Icons[Name]))
        if type(Icon) == "string" then return Icon end
        if type(Icon) == "table" then
            if type(Icon.Image) == "string" then return Icon.Image end
            if type(Icon.Image) == "number" then return "rbxassetid://" .. Icon.Image end
        end

        if Interface.Icons[Name] then return Interface.Icons[Name] end
        local Set = GetLucide()
        local Id = Set and (Set.byName[Name] or Set.byNorm[NormalizeIcon(Name)])
        return Id and ("rbxassetid://" .. Id) or nil
    end

    function Interface.ListIcons(Filter)
        local Set = GetLucide()
        local Result = {}
        if not Set then return Result end
        Filter = Filter and string.lower(Filter) or nil
        for Name in pairs(Set.byName) do
            if not Filter or string.find(Name, Filter, 1, true) then table.insert(Result, Name) end
        end
        table.sort(Result)
        return Result
    end

    function Interface.IconCount()
        local Set = GetLucide()
        return Set and Set.count or 0
    end

        local Success, g = pcall(function() return getfenv(0) end)

        local Success2, gg = pcall(function() return getgenv() end)
        if Success2 and type(gg) == "table" then table.insert(Environments, gg) end
        if type(_G) == "table" then table.insert(Environments, _G) end

    local function FindFunction(...)
        for _, Name in ipairs({ ... }) do
            for _, e in ipairs(Environments) do
                local Success, v = pcall(function() return e[Name] end)
                if Success and type(v) == "function" then return v end
            end
        end
        return nil
    end

    local function EnsureFolder(Path)
        if not (Executor.makefolder and Executor.isfolder) then return false end
        local PathPart
        for part in string.gmatch(Path, "[^/\\]+") do
            PathPart = PathPart and (PathPart .. "/" .. part) or part
            local Success, Exists = pcall(Executor.isfolder, PathPart)
            if Success and not Exists then pcall(Executor.makefolder, PathPart) end
        end
        return true
    end

    function Interface.SetIcon(Name, Source)
        if type(Source) ~= "string" then return false end
        if string.match(Source, "^rbxasset") then
            Interface.Icons[Name] = Source
            Interface.CustomIcons[Name] = Source
            return true
        end
        if string.match(Source, "^https?://") then
            if not (Executor.isfile and Executor.writefile and Executor.getcustomasset) then return false end
            EnsureFolder(Interface.IconFolder)
            local Path = Interface.IconFolder .. "/" .. string.gsub(tostring(Name), "[^%w_%-]", "") .. ".png"
            local Ok, Exists = pcall(Executor.isfile, Path)
            if not (Ok and Exists) then
                local Downloaded, Data = pcall(function() return game:HttpGet(Source) end)
                if not Downloaded or type(Data) ~= "string" or #Data == 0 then return false end
                if not pcall(Executor.writefile, Path, Data) then return false end
            end
            Source = Path
        end
        if not string.find(Source, "[/\\%.]") then
            local Id = Interface.Icon("lucide:" .. Source)
            if Id then Interface.Icons[Name] = Id Interface.CustomIcons[Name] = Id return true end
        end
        if not (Executor.getcustomasset and Executor.isfile) then return false end
        local Success, Exists = pcall(Executor.isfile, Source)
        if not (Success and Exists) then return false end
        local Success2, Id = pcall(Executor.getcustomasset, Source)
        if Success2 and type(Id) == "string" then
            Interface.Icons[Name] = Id
            Interface.CustomIcons[Name] = Id
            return true
        end
        return false
    end

    function Interface.UseLocalIcons(Folder)
        Folder = Folder or Interface.IconFolder
        if not (Executor.getcustomasset and Executor.isfile) then return 0 end
        local n = 0
        for Name in pairs(Interface.Icons) do
            if Interface.SetIcon(Name, Folder .. "/" .. Name .. ".png") then n += 1 end
        end
        return n
    end

    local function Create(ClassName, Properties, Children)
        local Object, Parent = Instance.new(ClassName), nil
        for k, v in pairs(Properties or {}) do
            if k == "Parent" then Parent = v else Object[k] = v end
        end
        for _, k in ipairs(Children or {}) do k.Parent = Object end
        Object.Parent = Parent
        return Object
    end

    function Interface:_trackPointer(Area, OnStart, OnMove, OnEnd)
        local Active

        self:_track(Area.InputBegan:Connect(function(Input)
            if Active or (Input.UserInputType ~= Enum.UserInputType.MouseButton1
                and Input.UserInputType ~= Enum.UserInputType.Touch) then return end

            Active = Input
            OnStart(Input.Position)
        end))

        self:_track(UserInputService.InputChanged:Connect(function(Input)
            if not Active then return end
            if Active.UserInputType == Enum.UserInputType.Touch then
                if Input ~= Active then return end
            elseif Input.UserInputType ~= Enum.UserInputType.MouseMovement then
                return
            end

            OnMove(Input.Position)
        end))

        self:_track(UserInputService.InputEnded:Connect(function(Input)
            if not Active or (Input ~= Active and not (Active.UserInputType == Enum.UserInputType.MouseButton1
                and Input.UserInputType == Enum.UserInputType.MouseButton1)) then return end
            Active = nil
            if OnEnd then OnEnd() end
        end))
    end

    function Interface:_makeDraggable(Wrap, Handle, OnTap, OnDetach)
        local StartPosition, StartPointer, Moved = Wrap.Position, nil, false

        self:_trackPointer(Handle, function(Position)
            StartPosition, StartPointer, Moved = Wrap.Position, Position, false
        end, function(Position)
            local Delta = Position - StartPointer
            local Distance = self.Touch and 8 or 4
            if not Moved and (math.abs(Delta.X) > Distance or math.abs(Delta.Y) > Distance) then
                Moved = true
                if OnDetach then OnDetach() end
            end

            if Moved then
                local Scale = self.Root.UIScale.Scale
                Wrap.Position = UDim2.fromOffset(
                    math.clamp(math.round(StartPosition.X.Offset + Delta.X / Scale), 0,
                        math.max(0, (self._usable or 1920) - Wrap.Size.X.Offset)),
                    math.clamp(math.round(StartPosition.Y.Offset + Delta.Y / Scale), 0,
                        math.max(0, (self._usableHeight or 1080) - Metrics.HEADER_SHUT)))
            end
        end, function()
            if not Moved and OnTap then OnTap() end
        end)
    end

    function Interface:_grabArea(Area, Apply, OnRelease)
        self:_trackPointer(Area, function(Position)
            Apply(Position.X, Position.Y)
        end, function(Position)
            Apply(Position.X, Position.Y)
        end, OnRelease)
    end

    function Interface.new(Config)
        Config = Config or {}
        local self = setmetatable({}, Interface)

        self.ThemeName    = Config.Theme or "Frost"
        self.Theme        = Interface.Themes[self.ThemeName] or Interface.Themes.Frost
        self.ToggleKey    = Config.ToggleKey or Enum.KeyCode.RightShift
        self.ShowBackdrop = Config.Backdrop == true

        if Config.HideTopbar then
            pcall(function() StarterGui:SetCore("TopbarEnabled", false) end)
        end

        self._refresh  = {}
        self._conns    = {}
        self._tabs     = {}
        self._tabOrder = {}
        self._controls = {}
        self._slots    = {}
        self._visible  = true
        self._binds    = {}
        self._listening = nil

        self.Touch = UserInputService.TouchEnabled
        self.OptionH = Metrics.OPTION_H + (self.Touch and 16 or 0)
        self.MenuButton = Config.MenuButton
        if self.MenuButton == nil then self.MenuButton = self.Touch end
        self._forceCompact = Config.Compact

        self._search   = Config.Search == true
        self._navTop   = Metrics.NAV_TOP + (self._search and 38 or 0)
        self._sidebarH = Metrics.SIDEBAR_H + (self._search and 38 or 0)

        self:_buildRoot()
        self:_buildSidebar(Config.Title or "Sense", Config.Mark)
        self:_wireGlobalInput()
        if self.MenuButton then
            self:_buildMenuButton(Config.Mark or string.sub(Config.Title or "Sense", 1, 1))
        end
        return self
    end

    function Interface:_track(Connection)
        table.insert(self._conns, Connection)
        return Connection
    end

    function Interface:_onRefresh(fn, Owner)
        table.insert(self._refresh, { fn = fn, owner = Owner or self._owner })
        fn(false)
        return fn
    end

    function Interface:_offRefresh(fn)
        for i = #self._refresh, 1, -1 do
            if self._refresh[i].fn == fn then table.remove(self._refresh, i) end
        end
    end

    local function IsWithin(inst, Root)
        return typeof(inst) == "Instance" and (inst == Root or inst:IsDescendantOf(Root))
    end

    function Interface:_forget(Root)
        for i = #self._refresh, 1, -1 do
            if IsWithin(self._refresh[i].owner, Root) then table.remove(self._refresh, i) end
        end
        for i = #self._binds, 1, -1 do
            local b = self._binds[i]
            if IsWithin(b.owner, Root) then
                if b.held and b.release then b.release() end
                table.remove(self._binds, i)
            end
        end
        for i = #self._controls, 1, -1 do
            if IsWithin(self._controls[i].owner, Root) then table.remove(self._controls, i) end
        end
        if self._popupRegion and IsWithin(self._popupRegion, Root) then
            self._closePopup, self._popupRegion = nil, nil
        end
        if self._listening and self._listeningOwner and IsWithin(self._listeningOwner, Root) then
            self._listening, self._listeningOwner = nil, nil
        end
    end

    function Interface:_buildRoot()
        local Gui = Create("ScreenGui", {
            Name = "F26Interface", IgnoreGuiInset = true, ResetOnSpawn = false,
            ZIndexBehavior = Enum.ZIndexBehavior.Sibling, DisplayOrder = 50,
        })
        self.Gui = Gui

        local Backdrop = Create("Frame", {
            Name = "Backdrop", BorderSizePixel = 0, ZIndex = 0,
            BackgroundColor3 = Color3.new(1, 1, 1),
            Size = UDim2.fromScale(1, 1), Visible = self.ShowBackdrop, Parent = Gui,
        })
        local Gradient = Create("UIGradient", { Rotation = 45, Parent = Backdrop })
        self.Backdrop = Backdrop
        self:_onRefresh(function()
            Gradient.Color = ColorSequence.new(self.Theme.backdropA, self.Theme.backdropB)
        end)

        local Layer = Create("Frame", {
            Name = "Layer", BackgroundTransparency = 1, ZIndex = 1,
            Size = UDim2.fromScale(1, 1), Parent = Gui,
        })
        self.Root = Layer

        local Scale = Create("UIScale", { Parent = Layer })
        local Camera
        local function Fit()
            Camera = workspace.CurrentCamera
            local Viewport = (Camera and Camera.ViewportSize) or Vector2.new(1920, 1080)
            if Viewport.X < 64 or Viewport.Y < 64 then return end

            if self.Touch then
                Scale.Scale = math.clamp(math.min(Viewport.X / (Metrics.PANEL_W + 24), Viewport.Y / 620), 0.9, 1.25)
            else
                Scale.Scale = math.clamp(Viewport.Y / 1080, 0.6, 1.6)
            end

            local AvailableWidth = Viewport.X / Scale.Scale
            local AvailableHeight = Viewport.Y / Scale.Scale
            self._usable, self._usableHeight = AvailableWidth, AvailableHeight

            if self._forceCompact == nil then
                self.Compact = AvailableWidth < (Metrics.WINDOW_X + Metrics.PANEL_W + Metrics.SIDEBAR_X)
            else
                self.Compact = self._forceCompact
            end

            if self.SidebarWrap then
                self._sidebarH = math.min(self._navTop + #self._tabOrder * Metrics.NAV_PITCH + 8,
                    math.max(self._navTop + 48, math.floor((self.Touch or self.Compact)
                        and AvailableHeight * 0.44 or AvailableHeight - Metrics.SIDEBAR_Y - 16)))
                self.SidebarWrap.Size = UDim2.fromOffset(Metrics.PANEL_W, self._sidebarH)
                if self.Compact then
                    self.SidebarWrap.Position = UDim2.fromOffset(
                        math.max(0, math.floor((AvailableWidth - Metrics.PANEL_W) / 2)), Metrics.SIDEBAR_Y)
                end
            end

            if self.NavList then
                local Height = self._sidebarH - self._navTop - 8
                self.NavHint.Visible = #self._tabOrder * Metrics.NAV_PITCH > Height
                self.NavList.Size = UDim2.new(1, 0, 0, Height - (self.NavHint.Visible and 18 or 0))
                self.NavList.CanvasSize = UDim2.fromOffset(0, #self._tabOrder * Metrics.NAV_PITCH)
            end

            local Top = self.Compact and (Metrics.SIDEBAR_Y + self._sidebarH + 10) or Metrics.WINDOW_Y
            self._maxWindowH = math.max(Metrics.HEADER_H + 60, math.floor(AvailableHeight - Top - 16))
            self:_refitWindows()
            self:_placeWindows()
        end

        self._refit = Fit
        Fit()
        task.defer(Fit)
        self:_track(workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
            if self._cameraConnection then self._cameraConnection:Disconnect() end
            Camera = workspace.CurrentCamera
            if Camera then
                self._cameraConnection = self:_track(Camera:GetPropertyChangedSignal("ViewportSize"):Connect(Fit))
            end
            Fit()
        end))
        if Camera then
            self._cameraConnection = self:_track(Camera:GetPropertyChangedSignal("ViewportSize"):Connect(Fit))
        end
    end

    function Interface:Open(Name)
        local Tab = self._tabs[Name]
        if not Tab or Tab.Open then return end

        if self.Touch then
            for i, v in pairs(self._tabs) do
                if v.Open then self:Close(i) end
            end
        end

        Tab.Open = true
        if not Tab.Detached then
            local i = 1
            while self._slots[i] do i += 1 end
            self._slots[i], Tab.Slot = Tab, i
        end
        Tab.Wrap.Visible = true
        self:_placeWindows()
        Tab._refreshNav(true)
    end

    function Interface:_refitWindows()
        for _, n in ipairs(self._tabOrder or {}) do
            local t = self._tabs[n]
            if t and t.Expanded then
                t.Wrap.Size = UDim2.fromOffset(Metrics.PANEL_W, t:Height())
            end
        end
    end

    function Interface:_placeWindows()
        if not self._slots then return end
        if self.Compact then
            local y = Metrics.SIDEBAR_Y + self._sidebarH + 10
            local x = self.Touch and math.max(0, math.floor(((self._usable or 1920) - Metrics.PANEL_W) / 2))
                or Metrics.SIDEBAR_X
            local Keys = {}
            for i in pairs(self._slots) do table.insert(Keys, i) end
            table.sort(Keys)
            for _, i in ipairs(Keys) do
                local Tab = self._slots[i]
                if Tab and not Tab.Detached then
                    Tab.Wrap.Position = UDim2.fromOffset(x, y)
                    y += Tab.Wrap.Size.Y.Offset + 10
                end
            end
            return
        end
        local Room = (self._usable or 1920) - Metrics.WINDOW_X
        local PerRow = math.max(1, math.floor(Room / Metrics.WINDOW_PITCH))
        for i, Tab in pairs(self._slots) do
            if Tab and not Tab.Detached then
                local Color = (i - 1) % PerRow
                local Row = (i - 1) // PerRow
                Tab.Wrap.Position = UDim2.fromOffset(
                    Metrics.WINDOW_X + Color * Metrics.WINDOW_PITCH,
                    Metrics.WINDOW_Y + Row * (Metrics.SIDEBAR_H + 10))
            end
        end
    end

    function Interface:Close(Name)
        local Tab = self._tabs[Name]
        if not Tab or not Tab.Open then return end
        Tab.Open = false
        self:_freeSlot(Tab)
        Tab.Wrap.Visible = false
        self:_placeWindows()
        Tab._refreshNav(true)
    end

    function Tab:Height()
        local Natural = Metrics.HEADER_H + (self._content or 0) + Metrics.CONTENT_BOT
        return math.max(Metrics.HEADER_SHUT, math.min(Natural, self.UI._maxWindowH or 10000))
    end

    function Tab:SetExpanded(Visible, Instant)
        self.Expanded = Visible
        local h = Visible and self:Height() or Metrics.HEADER_SHUT
        if Instant then
            self.Wrap.Size = UDim2.fromOffset(Metrics.PANEL_W, h)
            self.Chevron.Rotation = Visible and 0 or -90
        else
            PlayTween(self.Wrap, { Size = UDim2.fromOffset(Metrics.PANEL_W, h) }, MoveTween)
            PlayTween(self.Chevron, { Rotation = Visible and 0 or -90 }, MoveTween)
        end
        if self.UI.Compact then
            self.UI:_placeWindows()
            if not Instant then task.delay(0.2, function() self.UI:_placeWindows() end) end
        end
    end

    function Tab:_relayout()
        local y, Last = 5, nil
        for _, it in ipairs(self._items) do
            if it.hidden then
                it.frame.Visible = false
            else
                it.frame.Visible = true
                if Last then
                    y += (Last == "module" and it.kind == "module") and Metrics.MODULE_GAP or Metrics.GROUP_GAP
                end
                it.frame.Position = UDim2.fromOffset(0, y)
                y += it.height
                Last = it.kind
            end
        end
        self._content = y
        self.Content.CanvasSize = UDim2.fromOffset(0, y + Metrics.CONTENT_BOT)
        if self.Expanded then self.Wrap.Size = UDim2.fromOffset(Metrics.PANEL_W, self:Height()) end
        if self.UI.Compact then self.UI:_placeWindows() end
    end

    function Tab:_add(Kind, Height, Frame)
        self.UI._owner = Frame
        local Item = { kind = Kind, height = Height, frame = Frame }
        table.insert(self._items, Item)
        self:_relayout()
        return Item
    end

            Hovering = true
            Refresh(true)

            Hovering = false
            Refresh(true)
        end))
        UI:_track(Hitbox.Activated:Connect(function() Control:Set(not State) end))

        UI:_register(self.Name, Text, Control)
        return Control
    end

    function Group:_relayout()
        local y, Last = self.Header and Metrics.SECTION_H + Metrics.CTRL_GAP or 0, nil
        for _, it in ipairs(self._items) do
            it.frame.Visible = not it.hidden
            if not it.hidden then
                if Last then
                    y += (Last == "toggle" and it.kind == "toggle") and Metrics.TOGGLE_GAP or Metrics.CTRL_GAP
                end
                it.frame.Position = UDim2.fromOffset(Metrics.CTRL_X, y)
                y += it.height
                Last = it.kind
            end
        end
        self._y = y
        self.Holder.Size = UDim2.new(1, 0, 0, y)
        self.Bar.Size = UDim2.fromOffset(Metrics.ACCENT_W, y)
        self._item.height = y
        self.Tab:_relayout()
    end

    function Group:Clear()
        local UI = self.UI
        for _, it in ipairs(self._items) do
            UI:_forget(it.frame)
            it.frame:Destroy()
        end
        table.clear(self._items)
        self:_relayout()
    end

    function Tab:Remove()
        local UI = self.UI
        UI:Close(self.Name)
        UI:_forget(self.Wrap)
        UI:_forget(self.Nav)
        self.Wrap:Destroy()
        self.Nav:Destroy()
        UI._tabs[self.Name] = nil
        for i = #UI._tabOrder, 1, -1 do
            if UI._tabOrder[i] == self.Name then table.remove(UI._tabOrder, i) end
        end
        for i, n in ipairs(UI._tabOrder) do
            UI._tabs[n].Nav.Position = UDim2.fromOffset(
                Metrics.SIDEBAR_PAD, (UI.NavList and 0 or UI._navTop or Metrics.NAV_TOP)
                    + (i - 1) * Metrics.NAV_PITCH)
        end
        UI:_placeWindows()
        UI:_refit()
    end

    function Group:Toggle(Name, Default, Callback)
        local Control = { Kind = "toggle", Name = Name }
        local Row
        Row = self:_toggleRow(Name, Default and true or false, function()
            Row.setState(not Row.getState())
            if Callback then task.spawn(Callback, Row.getState()) end
        end)
        function Control:Get() return Row.getState() end
        function Control:Set(v, Silent)
            local NextValue = v and true or false
            if NextValue == Row.getState() then return end
            Row.setState(NextValue)
            if not Silent and Callback then task.spawn(Callback, Row.getState()) end
        end
        self.UI:_register(self.Tab.Name, Name, Control)
        return Control
    end

    function Group:Radio(names, Default, Callback)
        local Rows, Selected = {}, Default
        local Control = { Kind = "radio", Name = table.concat(names, "/") }
        local function apply(Choice, Silent)
            Selected = Choice
            for n, r in pairs(Rows) do r.setState(n == Choice) end
            if not Silent and Callback then task.spawn(Callback, Choice) end
        end
        for _, n in ipairs(names) do
            Rows[n] = self:_toggleRow(n, n == Default, function()
                if Selected ~= n then apply(n) end
            end)
            self._items[#self._items].filterName = Control.Name
        end
        function Control:Get() return Selected end
        function Control:Set(v, Silent)
            if Rows[v] and v ~= Selected then apply(v, Silent) end
        end
        self.UI:_register(self.Tab.Name, Control.Name, Control)
        return Control
    end

            ApplyProperties(BorderStroke, { Color = Theme.controlStroke }, a)

        UI:_track(Button.MouseEnter:Connect(function()
            Hovering = true
            Refresh(true)

        UI:_track(Button.MouseLeave:Connect(function()
            Hovering, Held = false, false
            Refresh(true)
        end))
        UI:_track(Button.MouseButton1Down:Connect(function()
            Held = true
            Refresh(false)
        end))
        UI:_track(Button.MouseButton1Up:Connect(function()
            Held = false
            Refresh(true)
        end))
        UI:_track(Button.Activated:Connect(function() if Callback then task.spawn(Callback) end end))

        local Control = { Kind = "button", Name = Text, Instance = Button }
        function Control:SetText(t) Button.Text = tostring(t) end
        return Control
    end

    function Group:Slider(Name, Options, Callback)
        Options = Options or {}
        local UI = self.UI
        local MinValue     = Options.Min or 0
        local MaxValue     = Options.Max or 1
        local Suffix   = Options.Suffix or ""
        local Decimals = Options.Decimals or 2
        local Step     = Options.Step
        local Value    = Options.Default or MinValue

        local Total = Metrics.SLIDER_LBL_H + Metrics.SLIDER_GAP + Metrics.TRACK_H + (UI.Touch and 16 or 0)
        local Holder = Create("Frame", {
            Name = Name, BackgroundTransparency = 1,
            Size = UDim2.fromOffset(Metrics.CTRL_W, Total), Parent = self.Holder,
        })
        self:_add("slider", Total, Holder)

        local NameLabel = Create("TextLabel", {
            BackgroundTransparency = 1, Text = Name, Font = BoldFont, TextSize = Metrics.TEXT_SM,
            TextXAlignment = Enum.TextXAlignment.Left,
            Size = UDim2.new(0.62, 0, 0, Metrics.SLIDER_LBL_H), Parent = Holder,
        })
        local ValueLabel = Create("TextLabel", {
            Name = "Value", BackgroundTransparency = 1, Text = "",
            Font = MediumFont, TextSize = Metrics.TEXT_SM,
            TextXAlignment = Enum.TextXAlignment.Right, AnchorPoint = Vector2.new(1, 0),
            Position = UDim2.fromScale(1, 0),
            Size = UDim2.new(0.38, 0, 0, Metrics.SLIDER_LBL_H), Parent = Holder,
        })
        local Track = Create("Frame", {
            Name = "Track", BorderSizePixel = 0,
            Position = UDim2.fromOffset(0, Metrics.SLIDER_LBL_H + Metrics.SLIDER_GAP + (UI.Touch and 8 or 0)),
            Size = UDim2.new(1, 0, 0, Metrics.TRACK_H), Parent = Holder,
        })
        SetCorner(Track, Metrics.TRACK_H // 2)
        local Fill = Create("Frame", { Name = "Fill", BorderSizePixel = 0, Parent = Track })
        SetCorner(Fill, Metrics.TRACK_H // 2)
        local Knob = Create("Frame", {
            Name = "Knob", BorderSizePixel = 0, AnchorPoint = Vector2.new(0.5, 0.5),
            Size = UDim2.fromOffset(Metrics.KNOB, Metrics.KNOB), Parent = Track,
        })
        SetCorner(Knob, Metrics.KNOB // 2)
        local KnobStroke = CreateStroke(Knob, Color3.new(), 2, 0)

        UI:_onRefresh(function(a)
            local Theme = UI.Theme
            ApplyProperties(NameLabel, { TextColor3 = Theme.text }, a)
            ApplyProperties(ValueLabel, { TextColor3 = Theme.textDim }, a)
            ApplyProperties(Track, { BackgroundColor3 = Theme.track }, a)
            ApplyProperties(Fill, { BackgroundColor3 = Theme.accent }, a)
            ApplyProperties(Knob, { BackgroundColor3 = Theme.panel }, a)
            ApplyProperties(KnobStroke, { Color = Theme.accent }, a)
        end)

        local Control = { Kind = "slider", Name = Name }
        local function Render()
            local a = (Value - MinValue) / (MaxValue - MinValue)
            Fill.Size = UDim2.fromScale(a, 1)
            Knob.Position = UDim2.new(a, 0, 0.5, 0)
            ValueLabel.Text = string.format("%." .. Decimals .. "f%s", Value, Suffix)
        end

        local OnRelease, Dragging = Options.Commit == "release", false

        function Control:Get() return Value end
        function Control:Set(v, Silent)
            if Step then v = math.round(v / Step) * Step end
            v = math.clamp(v, MinValue, MaxValue)
            if v == Value then return end
            Value = v
            Render()
            if Silent then return end
            if OnRelease and Dragging then return end
            if Callback then task.spawn(Callback, Value) end
        end
        Render()

        local Hitbox = Create("Frame", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 0, 0.5, UI.Touch and -16 or -Metrics.TRACK_H / 2),
            Size = UDim2.new(1, 0, 0, UI.Touch and 32 or Metrics.TRACK_H),
            Parent = Track,
        })

        UI:_grabArea(Hitbox, function(px)
            Dragging = true
            local a = math.clamp((px - Track.AbsolutePosition.X) / math.max(Track.AbsoluteSize.X, 1), 0, 1)
            Control:Set(MinValue + a * (MaxValue - MinValue))
        end, function()
            Dragging = false
            if OnRelease and Callback then task.spawn(Callback, Value) end
        end)

        UI:_register(self.Tab.Name, Name, Control)
        return Control
    end

            ApplyProperties(ListStroke, { Color = UI.Theme.moduleStroke }, a)
        end)

        local Selected = {}
        if Multi then
            for _, v in ipairs(Default or {}) do Selected[v] = true end
        elseif Default ~= nil then
            Selected[Default] = true
        end

        local Control = { Kind = Multi and "multidropdown" or "dropdown", Name = Name }
        function Control:Get()
            if Multi then
                local Result = {}
                for _, Object in ipairs(Options) do if Selected[Object] then table.insert(Result, Object) end end
                return Result
            end
            for _, Object in ipairs(Options) do if Selected[Object] then return Object end end
            return nil
        end

        local RefreshRow = {}
        local function UpdateValue()
            if Multi then
                local Picked = Control:Get()
                ValueLabel.Text = (#Picked == 0) and "None" or table.concat(Picked, ", ")
            else
                ValueLabel.Text = Control:Get() or "None"
            end
            for _, fn in ipairs(RefreshRow) do fn(true) end
        end

        local Open, SetOpen = false, nil
        local function ClosePopup() if Open then SetOpen(false) end end
        SetOpen = function(v)
            if Open == v then return end
            if v and UI._closePopup then UI._closePopup() end
            Open = v
            List.Visible = v
            Holder.Size = UDim2.fromOffset(Metrics.CTRL_W, v and OpenHeight or Metrics.HEAD_H)
            Item.height = v and OpenHeight or Metrics.HEAD_H
            Chevron.Rotation = v and 0 or -90
            self:_relayout()
            if v then UI._closePopup, UI._popupRegion = ClosePopup, Holder
            elseif UI._closePopup == ClosePopup then UI._closePopup, UI._popupRegion = nil, nil end
        end
        UI:_track(Header.Activated:Connect(function() SetOpen(not Open) end))

        local RowFrames = {}
        local function BuildRows()
            for i, Option in ipairs(Options) do
                local Row = Create("TextButton", {
                    Name = Option, Text = "", AutoButtonColor = false, BorderSizePixel = 0,
                    BackgroundTransparency = 1,
                    Position = UDim2.fromOffset(0, Metrics.BODY_PAD + (i - 1) * OptionHeight),
                    Size = UDim2.new(1, 0, 0, OptionHeight), Parent = List,
                })
                table.insert(RowFrames, Row)
                local Label = Create("TextLabel", {
                    BackgroundTransparency = 1, Text = Option, Font = MediumFont, TextSize = Metrics.TEXT_SM,
                    TextXAlignment = Enum.TextXAlignment.Left, AnchorPoint = Vector2.new(0, 0.5),
                    Position = UDim2.new(0, 10, 0.5, 0), Size = UDim2.new(1, -32, 0, 18), Parent = Row,
                })
                local Dot = Create("Frame", {
                    Name = "Dot", BorderSizePixel = 0, AnchorPoint = Vector2.new(1, 0.5),
                    Position = UDim2.new(1, -10, 0.5, 0), Size = UDim2.fromOffset(6, 6), Parent = Row,
                })
                SetCorner(Dot, 3)

                local Hovering = false
                local Refresh = UI:_onRefresh(function(a)
                    local Theme, Visible = UI.Theme, Selected[Option] == true
                    ApplyProperties(Label, { TextColor3 = Visible and Theme.accent or (Hovering and Theme.text or Theme.textDim) }, a)
                    ApplyProperties(Dot, { BackgroundColor3 = Theme.accent, BackgroundTransparency = Visible and 0 or 1 }, a)
                    ApplyProperties(Row, { BackgroundColor3 = Theme.text, BackgroundTransparency = Hovering and 0.96 or 1 }, a)
                end, Row)
                table.insert(RefreshRow, Refresh)

                UI:_track(Row.MouseEnter:Connect(function()

                    Hovering = true

                    Refresh(true)

                end))
                UI:_track(Row.MouseLeave:Connect(function()
                    Hovering = false
                    Refresh(true)
                end))
                UI:_track(Row.Activated:Connect(function()
                    if Multi then
                        Selected[Option] = not Selected[Option] or nil
                    else
                        Selected = { [Option] = true }
                        SetOpen(false)
                    end
                    UpdateValue()
                    if Callback then task.spawn(Callback, Control:Get()) end
                end))
            end
        end
        BuildRows()

        function Control:SetOptions(NewOptions, PreserveSelection)
            for _, r in ipairs(RowFrames) do r:Destroy() end
            for _, fn in ipairs(RefreshRow) do UI:_offRefresh(fn) end
            table.clear(RowFrames)
            table.clear(RefreshRow)

            Options = NewOptions or {}
            if not PreserveSelection then Selected = {} end
            BodyHeight = Metrics.BODY_PAD * 2 + #Options * OptionHeight
            OpenHeight = Metrics.HEAD_H + Metrics.HEAD_GAP + BodyHeight
            List.Size = UDim2.new(1, 0, 0, BodyHeight)
            BuildRows()
            if Open then
                Holder.Size = UDim2.fromOffset(Metrics.CTRL_W, OpenHeight)
                Item.height = OpenHeight
                Group:_relayout()
            end
            UpdateValue()
        end

        function Control:Set(v, Silent)
            local NextValue = {}
            if Multi then
                if type(v) == "table" then
                    for _, Object in ipairs(v) do NextValue[Object] = true end
                end
            elseif v ~= nil then
                NextValue[v] = true
            end
            local Same = true
            for k in pairs(NextValue) do if not Selected[k] then Same = false break end end
            if Same then
                for k in pairs(Selected) do if not NextValue[k] then Same = false break end end
            end
            if Same then return end
            Selected = NextValue
            UpdateValue()
            if not Silent and Callback then task.spawn(Callback, Control:Get()) end
        end
        Control.Close = ClosePopup
        UpdateValue()

        UI:_register(self.Tab.Name, Name, Control)
        return Control
    end

            commit(Box.Text)
        end))

        function Control:Get() return Value end
        function Control:Set(v, Silent) commit(tostring(v or ""), Silent) end

        UI:_register(self.Tab.Name, Name, Control)
        return Control
    end

    function Group:ColorPicker(Name, Default, Options, Callback)

            Name = "Hex", BorderSizePixel = 0,

            Name = "Box", BackgroundTransparency = 1, Text = "", ClearTextOnFocus = false,
            Font = MediumFont, TextSize = Metrics.TEXT_SM, TextXAlignment = Enum.TextXAlignment.Left,
            AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 8, 0.5, 0),
            Size = UDim2.new(1, -60, 0, 16), Parent = HexField,

            Name = "Alpha", BackgroundTransparency = 1, Text = "", Font = MediumFont,
            TextSize = Metrics.TEXT_SM, TextXAlignment = Enum.TextXAlignment.Right,
            AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -8, 0.5, 0),
            Size = UDim2.fromOffset(48, 16), Visible = WithAlpha, Parent = HexField,

            local Theme = UI.Theme

            ApplyProperties(PercentLabel, { TextColor3 = Theme.textDim }, a)
        end)

        local StartColor, StartAlpha = Default, Options.DefaultAlpha
        if type(Default) == "table" then StartColor, StartAlpha = Default.c, Default.a end
        local h, s, v = Color3.toHSV(typeof(StartColor) == "Color3" and StartColor
            or Color3.fromRGB(240, 110, 25))
        local Alpha = WithAlpha and math.clamp(tonumber(StartAlpha) or 1, 0, 1) or 1
        local Control = { Kind = "color", Name = Name }

        local function ToHex(c)
            return string.format("#%02X%02X%02X",
                math.round(c.R * 255), math.round(c.G * 255), math.round(c.B * 255))
        end

        local function Render()
            local c = Color3.fromHSV(h, s, v)
            sv.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
            Cursor.Position = UDim2.fromScale(s, 1 - v)
            HueKnob.Position = UDim2.new(h, 0, 0.5, 0)
            Swatch.BackgroundColor3 = c
            Swatch.BackgroundTransparency = 1 - Alpha
            if WithAlpha then
                AlphaFill.BackgroundColor3 = c
                AlphaKnob.Position = UDim2.new(Alpha, 0, 0.5, 0)
                PercentLabel.Text = string.format("%d%%", math.round(Alpha * 100))
            end
            if not HexFocused then HexBox.Text = ToHex(c) end
        end

        function Control:Get() return Color3.fromHSV(h, s, v) end
        function Control:GetAlpha() return Alpha end

        local function apply(NewHue, NewSaturation, NextValue, na, Silent)
            na = WithAlpha and math.clamp(na, 0, 1) or 1
            if NewHue == h and NewSaturation == s and NextValue == v and na == Alpha then return end
            h, s, v, Alpha = NewHue, NewSaturation, NextValue, na
            Render()
            if not Silent and Callback then task.spawn(Callback, Control:Get(), Alpha) end
        end

        function Control:Set(c, Silent)
            local na = Alpha
            if type(c) == "table" then c, na = c.c, (c.a or Alpha) end
            if typeof(c) ~= "Color3" then return end
            local NewHue, NewSaturation, NextValue = Color3.toHSV(c)
            apply(NewHue, NewSaturation, NextValue, na, Silent)
        end
        function Control:SetAlpha(na, Silent) apply(h, s, v, tonumber(na) or Alpha, Silent) end

        Control.Encode = function()
            local c = Control:Get()
            local ColorText = string.format("%d,%d,%d",
                math.round(c.R * 255), math.round(c.G * 255), math.round(c.B * 255))
            return WithAlpha and (ColorText .. string.format(",%.3f", Alpha)) or ColorText
        end
        Control.Decode = function(Text)
            local r, g, bl, al = string.match(tostring(Text), "^(%d+),(%d+),(%d+),?([%d%.]*)$")
            if not r then return nil end
            return { c = Color3.fromRGB(tonumber(r), tonumber(g), tonumber(bl)), a = tonumber(al) }
        end

        local function ParseHex(t)
            t = string.gsub(string.gsub(t, "%s", ""), "^#", "")
            if not string.match(t, "^%x+$") then return nil end
            if #t == 3 then
                t = string.gsub(t, "(%x)", "%1%1")
            end
            if #t ~= 6 and #t ~= 8 then return nil end
            local c = Color3.fromRGB(tonumber(string.sub(t, 1, 2), 16),
                tonumber(string.sub(t, 3, 4), 16), tonumber(string.sub(t, 5, 6), 16))
            local a8 = (#t == 8) and tonumber(string.sub(t, 7, 8), 16) / 255 or nil
            return c, a8
        end

        UI:_track(HexBox.Focused:Connect(function()

            HexFocused = true

            HexRefresh(true)

        end))
        UI:_track(HexBox.FocusLost:Connect(function()
            HexFocused = false
            HexRefresh(true)
            local c, a8 = ParseHex(HexBox.Text)
            if c then
                local NewHue, NewSaturation, NextValue = Color3.toHSV(c)
                apply(NewHue, NewSaturation, NextValue, a8 or Alpha)
            end
            Render()
        end))

        Render()

        local Open, SetOpen = false, nil
        local function ClosePopup() if Open then SetOpen(false) end end
        SetOpen = function(Value)
            if Open == Value then return end
            if Value and UI._closePopup then UI._closePopup() end
            Open = Value
            Body.Visible = Value
            Holder.Size = UDim2.fromOffset(Metrics.CTRL_W, Value and OpenHeight or Metrics.HEAD_H)
            Item.height = Value and OpenHeight or Metrics.HEAD_H
            Chevron.Rotation = Value and 0 or -90
            self:_relayout()
            if Value then UI._closePopup, UI._popupRegion = ClosePopup, Holder
            elseif UI._closePopup == ClosePopup then UI._closePopup, UI._popupRegion = nil, nil end
        end
        UI:_track(Header.Activated:Connect(function() SetOpen(not Open) end))
        Control.Close = ClosePopup

        local function Fraction(Frame, p, axis)
            local Position, Size = Frame.AbsolutePosition, Frame.AbsoluteSize
            if axis == "Y" then return math.clamp((p - Position.Y) / math.max(Size.Y, 1), 0, 1) end
            return math.clamp((p - Position.X) / math.max(Size.X, 1), 0, 1)
        end
        UI:_grabArea(sv, function(px, py)
            apply(h, Fraction(sv, px), 1 - Fraction(sv, py, "Y"), Alpha)
        end)
        UI:_grabArea(Hue, function(px) apply(Fraction(Hue, px), s, v, Alpha) end)
        if WithAlpha then
            UI:_grabArea(AlphaBar, function(px) apply(h, s, v, Fraction(AlphaBar, px)) end)
        end

        UI:_register(self.Tab.Name, Name, Control)
        return Control
    end

    local MODS = {
        Ctrl  = { Enum.KeyCode.LeftControl, Enum.KeyCode.RightControl },
        Shift = { Enum.KeyCode.LeftShift, Enum.KeyCode.RightShift },
        Alt   = { Enum.KeyCode.LeftAlt, Enum.KeyCode.RightAlt },
    }
    local MOD_ORDER = { "Ctrl", "Shift", "Alt" }
    local MOUSE_KEYS = {
        [Enum.UserInputType.MouseButton2] = "MB2",
        [Enum.UserInputType.MouseButton3] = "MB3",
    }

    local function EmptyModifiers() return { Ctrl = false, Shift = false, Alt = false } end

    local function IsModifier(k)
        for _, pair in pairs(MODS) do
            if k == pair[1] or k == pair[2] then return true end
        end
        return false
    end

    local function HeldModifiers()
        local Result = EmptyModifiers()
        for Name, pair in pairs(MODS) do
            Result[Name] = UserInputService:IsKeyDown(pair[1]) or UserInputService:IsKeyDown(pair[2])
        end
        return Result
    end

    local function InputKey(Input)
        if MOUSE_KEYS[Input.UserInputType] then return Input.UserInputType end
        if Input.KeyCode ~= Enum.KeyCode.Unknown then return Input.KeyCode end
        return nil
    end

    local function KeyName(k)
        return MOUSE_KEYS[k] or k.Name
    end

    local function KeyFromName(n)
        if n == "" or n == "None" or n == "Unknown" then return nil end
        for Input, Label in pairs(MOUSE_KEYS) do
            if Label == n then return Input end
        end
        local Success, k = pcall(function() return Enum.KeyCode[n] end)
        if Success and k and k ~= Enum.KeyCode.Unknown then return k end
        return nil
    end

    function Group:Keybind(Name, Default, Callback, Options)
        if type(Callback) == "table" and Options == nil then Callback, Options = nil, Callback end
        Options = Options or {}
        local UI = self.UI

        local Row = Create("TextButton", {
            Name = Name, Text = "", AutoButtonColor = false, BorderSizePixel = 0,
            Size = UDim2.fromOffset(Metrics.CTRL_W, Metrics.HEAD_H), Parent = self.Holder,
        })
        self:_add("control", Metrics.HEAD_H, Row)
        SetCorner(Row, Metrics.R_CTRL)
        local RowStroke = CreateStroke(Row, Color3.new(), 1, 0.25)

        local NameLabel = Create("TextLabel", {
            BackgroundTransparency = 1, Text = Name, Font = BoldFont, TextSize = Metrics.TEXT_SM,
            TextXAlignment = Enum.TextXAlignment.Left, AnchorPoint = Vector2.new(0, 0.5),
            Position = UDim2.new(0, 10, 0.5, 0), Size = UDim2.new(0.5, 0, 0, 18), Parent = Row,
        })
        local KeyLabel = Create("TextLabel", {
            Name = "Key", BackgroundTransparency = 1, Text = "", Font = MediumFont,
            TextSize = Metrics.TEXT_SM, TextTruncate = Enum.TextTruncate.AtEnd,
            TextXAlignment = Enum.TextXAlignment.Right, AnchorPoint = Vector2.new(1, 0.5),
            Position = UDim2.new(1, -10, 0.5, 0), Size = UDim2.new(0.5, -10, 0, 18), Parent = Row,
        })

        local Key, Modifiers = nil, EmptyModifiers()
        local Mode = (Options.Mode == "Hold") and "Hold" or "Toggle"
        local Listening, Hovering, BoundAt = false, false, 0
        local Control = { Kind = "keybind", Name = Name }

        local function Combo()
            if not Key then return "None" end
            local Parts = {}
            for _, m in ipairs(MOD_ORDER) do if Modifiers[m] then table.insert(Parts, m) end end
            table.insert(Parts, KeyName(Key))
            return table.concat(Parts, "+")
        end

        local Refresh = UI:_onRefresh(function(a)
            local Theme = UI.Theme
            ApplyProperties(Row, { BackgroundColor3 = Hovering and Theme.control:Lerp(Theme.text, 0.06) or Theme.control }, a)
            ApplyProperties(RowStroke, { Color = Listening and Theme.accent or Theme.controlStroke,
                Transparency = Listening and 0 or 0.25 }, a)
            ApplyProperties(NameLabel, { TextColor3 = Theme.text }, a)
            KeyLabel.Text = Listening and "..." or (Combo() .. ((Mode == "Hold" and Key) and "  hold" or ""))
            ApplyProperties(KeyLabel, { TextColor3 = Listening and Theme.accent or Theme.textDim }, a)
        end)

        local Binding = { owner = Row, held = false }
        Binding.matches = function(Input)
            if not Key or InputKey(Input) ~= Key then return false end
            local h = HeldModifiers()
            for _, m in ipairs(MOD_ORDER) do
                if h[m] ~= Modifiers[m] then return false end
            end
            return true
        end
        Binding.releases = function(Input) return Key ~= nil and InputKey(Input) == Key end
        Binding.press = function()
            if Mode == "Hold" then Binding.held = true end
            if Callback then Callback(true, Key) end
        end
        Binding.release = function()
            Binding.held = false
            if Callback then Callback(false, Key) end
        end
        table.insert(UI._binds, Binding)

        local function SetBinding(k, m, Silent)
            m = m or EmptyModifiers()
            if k == Key and m.Ctrl == Modifiers.Ctrl and m.Shift == Modifiers.Shift and m.Alt == Modifiers.Alt then
                Refresh(true)
                return
            end
            if Binding.held then Binding.release() end
            Key = k
            Modifiers = { Ctrl = m.Ctrl == true, Shift = m.Shift == true, Alt = m.Alt == true }
            Refresh(true)
            if not Silent and Control.OnBound then task.spawn(Control.OnBound, Key, Modifiers) end
        end

        local Listener
        local function StopListening()
            Listening = false
            if UI._listening == Listener then UI._listening, UI._listeningOwner = nil, nil end
            Refresh(true)
        end
        Listener = function(Input)
            local t = Input.UserInputType
            if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
                StopListening()
                return true
            end
            local k = InputKey(Input)
            if not k then return false end
            if k == Enum.KeyCode.Escape or k == Enum.KeyCode.Backspace then
                StopListening()
                SetBinding(nil, nil)
                return true
            end
            if IsModifier(k) then return true end
            local m = HeldModifiers()
            StopListening()
            BoundAt = os.clock()
            SetBinding(k, m)
            return true
        end

        UI:_track(Row.MouseEnter:Connect(function()

            Hovering = true

            Refresh(true)

        end))
        UI:_track(Row.MouseLeave:Connect(function()
            Hovering = false
            Refresh(true)
        end))
        UI:_track(Row.Activated:Connect(function()
            if Listening then return end

            if UI._listening then UI._listening({ UserInputType = Enum.UserInputType.MouseButton1 }) end
            Listening = true
            UI._listening, UI._listeningOwner = Listener, Row
            Refresh(true)
        end))
        UI:_track(Row.MouseButton2Click:Connect(function()

            if Listening or os.clock() - BoundAt < 0.3 then return end
            Control:SetMode(Mode == "Hold" and "Toggle" or "Hold")
        end))

        function Control:GetMode() return Mode end
        function Control:SetMode(m)
            m = (m == "Hold") and "Hold" or "Toggle"
            if m == Mode then return end
            if Binding.held then Binding.release() end
            Mode = m
            Refresh(true)
        end

        function Control:Get()
            if not Key then return "None" end
            return (Mode == "Hold" and "Hold+" or "") .. Combo()
        end
        function Control:Set(v, Silent)
            if typeof(v) == "EnumItem" then
                SetBinding((v ~= Enum.KeyCode.Unknown) and v or nil, nil, Silent)
                return
            end
            local m, k, md = EmptyModifiers(), nil, "Toggle"
            for part in string.gmatch(tostring(v or ""), "[^+]+") do
                if part == "Hold" then md = "Hold"
                elseif m[part] ~= nil then m[part] = true
                else k = KeyFromName(part) end
            end
            Control:SetMode(md)
            SetBinding(k, m, Silent)
        end

        if typeof(Default) == "EnumItem" then
            Key = (Default ~= Enum.KeyCode.Unknown) and Default or nil
        elseif type(Default) == "string" then
            Control:Set(Default, true)
        end
        Refresh(false)

        UI:_register(self.Tab.Name, Name, Control)
        return Control
    end

    function Interface:SetFilter(Query)
        Query = string.lower(tostring(Query or ""))
        self.Filter = Query
        for _, TabName in ipairs(self._tabOrder) do
            local t = self._tabs[TabName]
            local ShowAll = Query == "" or string.find(string.lower(TabName), Query, 1, true) ~= nil
            local function Matches(n)
                return ShowAll or string.find(string.lower(n), Query, 1, true) ~= nil
            end
            for _, it in ipairs(t._items) do
                if it.kind == "module" then
                    it.hidden = not Matches(it.frame.Name)
                elseif it.group then
                    local SectionMatch = ShowAll or it.group.Name and Matches(it.group.Name)
                    local Any = false
                    for _, c in ipairs(it.group._items) do
                        c.hidden = not (SectionMatch or Matches(c.filterName or c.frame.Name))
                        if not c.hidden then Any = true end
                    end
                    it.hidden = not SectionMatch and not Any
                    it.group:_relayout()
                end
            end
            t:_relayout()
        end
    end

    function Interface:Apply(Data)
        if type(Data) ~= "table" then return false end
        if Data.theme then self:SetTheme(Data.theme, true) end
        if Data.values then
            for _, e in ipairs(self._controls) do
                local c, Value = e.ctrl, Data.values[e.id]
                if Value ~= nil and c.Decode then Value = c.Decode(Value) end
                if Value ~= nil then pcall(function() c:Set(Value) end) end
            end
        end

        if type(Data.layout) == "table" then
            for n, l in pairs(Data.layout) do
                local t = self._tabs[self.TabAliases and self.TabAliases[n] or n]
                if t and type(l) == "table" then
                    t.Detached = l.detached and true or false
                    t:SetExpanded(l.expanded ~= false, true)
                end
            end
        end
        if Data.open then
            for _, n in ipairs(self._tabOrder) do self:Close(n) end
            for _, n in ipairs(Data.open) do self:Open(self.TabAliases and self.TabAliases[n] or n) end
        end

        if type(Data.layout) == "table" then
            for n, l in pairs(Data.layout) do
                local t = self._tabs[self.TabAliases and self.TabAliases[n] or n]
                if t and t.Detached and type(l) == "table" then
                    t.Wrap.Position = UDim2.fromOffset(l.x or Metrics.WINDOW_X, l.y or Metrics.WINDOW_Y)
                end
            end
        end
        return true
    end

local Force = function(Object)
    if Object:IsA("BodyForce") then return Object.Force end
    if not Object:IsA("VectorForce") or not Object.Enabled then return end
    if not Object.Attachment0 then return end
    local Attachment = Object.RelativeTo == Enum.ActuatorRelativeTo.Attachment0 and Object.Attachment0
        or Object.RelativeTo == Enum.ActuatorRelativeTo.Attachment1 and Object.Attachment1
    if Object.RelativeTo == Enum.ActuatorRelativeTo.World then return Object.Force end
    return Attachment and Attachment.WorldCFrame:VectorToWorldSpace(Object.Force)
end
local FutureForce = function(Model, State, Time)
    local Value = Model.Value
    if Model.Curve then
        local Duration = State.Duration
        local Switch = Duration / 1.3
        local Age = State.Time - Model.Created + Time
        if State.Flat and Age >= Duration * 3 then return Vector3.zero end
        if State.Reverse and not Model.Flipped and Age >= Switch then
            local Start = Model.Initial * Vector3.new(-1, 1, -1)
            return Start:Lerp(Start * Vector3.new(0.25, -0.5, 0.25), math.clamp((Age - Switch) / (Duration - Switch), 0, 1))
        end
        local Target = Model.Initial * Vector3.new(0.25, not Model.Flipped and State.Args.direction == "none" and -0.25 or -0.5, 0.25)
        if Model.Rate.Magnitude > 0.001 then
            local Remaining = (Target - Value):Dot(Model.Rate) / Model.Rate:Dot(Model.Rate)
            return Value + Model.Rate * math.min(Time, math.max(Remaining, 0))
        end
        return Value
    end
    return Value + Model.Rate * math.min(Time, 0.1)
end
local Forecast = function(Ball, State, Params)
    local Position, Velocity = Ball.Position, Ball.AssemblyLinearVelocity
    local Radius, Mass = math.min(Ball.Size.X, Ball.Size.Y, Ball.Size.Z) / 2, Ball.AssemblyMass
    local Points, Step = { Position }, Prediction.Time / 48
    local Ground = workspace:Raycast(Position, Vector3.new(0, -Radius - 0.1, 0), Params)
    Ground = Ground and Velocity.Y <= 1 and Ground.Normal.Y > 0.5 and Ground or nil
    if Ground then Position = Ground.Position + Ground.Normal * (Radius + 0.02) end
    for i = 1, 48 do
        local Time, Acceleration = (i - 0.5) * Step, Vector3.new(0, -workspace.Gravity, 0)
        for Object, Model in pairs(State.Forces) do
            if Model.Value then Acceleration += FutureForce(Model, State, Time) / Mass end
        end
        Acceleration += State.Residual * math.exp(-Time / 0.25)
        if Ground then
            Velocity -= Ground.Normal * math.min(Velocity:Dot(Ground.Normal), 0)
            Acceleration -= Ground.Normal * math.min(Acceleration:Dot(Ground.Normal), 0)
        end
        local NextVelocity = Velocity + Acceleration * Step
        for Object, Model in pairs(State.Forces) do
            if Model.Controller and State.Time + Time - Model.Created < 0.3 then
                local Change = (Object.Velocity - NextVelocity) * (1 - math.exp(-Object.P * Step / Mass))
                local Limit = Object.MaxForce * (Step / Mass)
                NextVelocity += Vector3.new(math.clamp(Change.X, -Limit.X, Limit.X),
                    math.clamp(Change.Y, -Limit.Y, Limit.Y), math.clamp(Change.Z, -Limit.Z, Limit.Z))
            end
        end
        if Ground then NextVelocity -= Ground.Normal * math.min(NextVelocity:Dot(Ground.Normal), 0) end
        local Move = (Velocity + NextVelocity) * (Step / 2)
        if Move.Magnitude < 0.001 then break end
        local Hit = workspace:Spherecast(Position, Radius, Move, Params)
        if Hit and not (Ground and Hit.Normal.Y > 0.5) then
            Points[#Points + 1] = Position + Move.Unit * math.max(Hit.Distance - 0.01, 0)
            break
        end
        Position, Velocity = Position + Move, NextVelocity
        Points[#Points + 1] = Position
        if Ground then
            Ground = workspace:Raycast(Position, Vector3.new(0, -Radius - 0.15, 0), Params)
            Ground = Ground and Ground.Normal.Y > 0.5 and Ground or nil
        end
    end
    return Points
end

local LastCatch = -math.huge
local AutoCatch = function(Ball)
    if not Dive.Catch or not Ball or os.clock() - LastCatch < 0.65 then return end
    local Character = Player.Character
    local Part = Character and Character:FindFirstChild("HumanoidRootPart")
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local Core = Root.toolData and Root.toolData.GK and Root.toolData.GK.activeController
    local Bind = Core and Core.equipped and Core.binds and Core.binds.catch
    if not Part or not Humanoid or Humanoid.Health <= 0 or Root.character ~= Character
        or not Bind or Bind.using or Root:isHolding() or RootScript:GetAttribute("forceDisable")
        or RootScript:GetAttribute("charging") or RootScript:GetAttribute("catchCooldown")
        or RootScript:GetAttribute("ragdoll") or RootScript:GetAttribute("globalOverride") and not RootScript:GetAttribute("diving")
        or game:GetService("UserInputService"):GetFocusedTextBox() then return end
    local Delta = Ball.Position - Part.Position
    if Delta.Magnitude > 14 then return end
    local Velocity = Ball.AssemblyLinearVelocity - Part.AssemblyLinearVelocity
    local Time = Velocity.Magnitude > 1 and math.clamp(-Delta:Dot(Velocity) / Velocity:Dot(Velocity), 0, 0.25) or 0
    if (Delta + Velocity * Time).Magnitude > 4.5 + math.min(Ball.Size.X, Ball.Size.Y, Ball.Size.Z) / 2
        or not Root:gkCheck() then return end
    LastCatch = os.clock()
    task.spawn(function()
        if Dive.Catch and Player.Character == Character and Root.character == Character
            and Root.toolData.GK.activeController == Core and Core.equipped and not Bind.using
            and Ball.Parent and not Ball.Anchored and not Ball:GetAttribute("catch")
            and not Root:isHolding() and not RootScript:GetAttribute("forceDisable") then
            Bind:inputBegan(Root)
        end
    end)
end

local LastDive = -math.huge
local AutoDive = function(Ball, Points)
    if not Dive.Enabled or os.clock() - LastDive < 1.8 then return end
    local Character = Player.Character
    local Part = Character and Character:FindFirstChild("HumanoidRootPart")
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local Core = Root.toolData and Root.toolData.GK and Root.toolData.GK.activeController
    if not Part or not Humanoid or Humanoid.Health <= 0 or Root.character ~= Character
        or not Core or not Core.equipped or not Core.binds or Root:isHolding()
        or RootScript:GetAttribute("globalOverride") or RootScript:GetAttribute("charging")
        or RootScript:GetAttribute("ragdoll") or RootScript:GetAttribute("forceDisable") then return end
    local Input = game:GetService("UserInputService")
    if Input:GetFocusedTextBox() then return end
    local Step = Prediction.Time / 48
    local Motion = Part.AssemblyLinearVelocity * Vector3.new(1, 0, 1)
    local Depth = 0.75 + math.min(Ball.Size.X, Ball.Size.Y, Ball.Size.Z) / 2
    local Previous = Part.CFrame:PointToObjectSpace(Points[1])
    for i = 2, #Points do
        local Current = Part.CFrame:PointToObjectSpace(Points[i] - Motion * ((i - 1) * Step))
        if Previous.Z < -Depth and Current.Z >= -Depth then
            local Alpha = (-Depth - Previous.Z) / (Current.Z - Previous.Z)
            local Position = Previous:Lerp(Current, Alpha)
            local Time = (i - 2 + Alpha) * Step
            local High = Position.Y > 0.75
            if math.abs(Position.X) > (High and 12 or 10) or Position.Y < -4 or Position.Y > 7 then return end
            local Lead = math.clamp(math.max(0.11, (math.abs(Position.X) - 2.5) / 23,
                High and (Position.Y - 2.5) / 12 or 0) + Dive.Lead, 0.15, 0.45)
            if Time > Lead then return end
            local Direction = math.abs(Position.X) <= 1.5 and "straight" or Position.X > 0 and "right" or "left"
            if Input:IsKeyDown(Enum.KeyCode.D) and Direction ~= "right"
                or Input:IsKeyDown(Enum.KeyCode.A) and Direction == "straight" then return end
            local Bind = Core.binds[High and "high dive" or "low dive"]
            if not Bind or Bind.using or not Root:gkCheck() then return end
            LastDive = os.clock()
            task.spawn(function()
                if Dive.Enabled and Player.Character == Character and Root.character == Character
                    and Root.toolData.GK.activeController == Core and Core.equipped
                    and Ball.Parent and not Ball.Anchored and not Ball:GetAttribute("catch") and not Root:isHolding()
                    and not Bind.using and not RootScript:GetAttribute("globalOverride")
                    and not RootScript:GetAttribute("forceDisable") then
                    Bind:inputBegan(Root, Direction)
                end
            end)
            return
        end
        Previous = Current
    end
end

local PredictionBall, Sample, LastPrediction
local Cast = RaycastParams.new()
Cast.FilterType, Cast.RespectCanCollide = Enum.RaycastFilterType.Exclude, true
Env.Predict = game:GetService("RunService").Heartbeat:Connect(function()
    local Ball, Now = PredictionBall, os.clock()
    if not (Prediction.Enabled or Dive.Enabled) or not Ball or not Ball.Parent or Ball.Anchored or Ball:GetAttribute("catch") then
        HidePath()
        Sample = nil
        return
    end
    if LastPrediction and Now - LastPrediction < 1 / 15 then return end
    LastPrediction = Now
    local Velocity = Ball.AssemblyLinearVelocity
    if Velocity.Magnitude < 1 then HidePath() Sample = nil return end
    if not Sample or Sample.Ball ~= Ball then
        Sample = { Ball = Ball, Velocity = Velocity, Time = Now, Args = {}, Forces = {} }
    end
    local Elapsed = Now - Sample.Time
    local Properties = Ball:FindFirstChild("properties")
    local Args = Properties and Properties:FindFirstChild("args")
    if Args ~= Sample.Folder then Sample.Args, Sample.Folder, Sample.Loaded = {}, Args, false end
    if Args and not Sample.Loaded and Args:FindFirstChild("loaded") then
        for i, v in pairs(Args:GetChildren()) do
            if v:IsA("ValueBase") then Sample.Args[v.Name] = v.Value end
        end
        Sample.Loaded = true
    end
    local Shot = Sample.Args
    local Launch = typeof(Shot.vector) == "Vector3" and Shot.vector or Velocity
    local Speed = Vector3.new(Launch.X, 0, Launch.Z).Magnitude
    Sample.Flat = Launch.Y <= 2 and not Shot.simpleFinesseCurve
    Sample.Duration = math.max(Speed / (Sample.Flat and 220 or Shot.risingShot and 40 or 120), 0.05)
    Sample.Knuckle = Shot.spinType == "knuckle"
    Sample.Reverse = Sample.Knuckle and (not Shot.lookAngleAtRelease or Shot.lookAngleAtRelease > 10)
    local Active, Total, Controller = {}, Vector3.zero, false
    for i, v in pairs(Ball:GetChildren()) do
        local Value = Force(v)
        if Value or v:IsA("BodyVelocity") then
            local Previous = Sample.Forces[v]
            local Model = {
                Value = Value, Created = Previous and Previous.Created or Now,
                Initial = Previous and Previous.Initial or Value,
                Rate = Value and Previous and Previous.Value and Elapsed > 0 and (Value - Previous.Value) / Elapsed or Vector3.zero,
                Curve = v:IsA("BodyForce") and Shot.curve == true,
                Controller = v:IsA("BodyVelocity"),
                Flipped = Previous and Previous.Flipped or false,
            }
            if Value then
                Total += Value
                Sample.Knuckle = Sample.Knuckle or v.Name:lower() == "knuckle" and Value.Magnitude > 0.01
                if Model.Initial and Vector3.new(Value.X, 0, Value.Z):Dot(Vector3.new(Model.Initial.X, 0, Model.Initial.Z)) < 0 then
                    Model.Flipped, Model.Initial = true, Value
                end
            end
            Active[v], Controller = Model, Controller or Model.Controller
        end
    end
    local Acceleration = Vector3.new(0, -workspace.Gravity, 0) + Total / Ball.AssemblyMass
    local Residual = Elapsed > 0 and (Velocity - Sample.Velocity) / Elapsed - Acceleration or Vector3.zero
    Sample.Residual = not Controller and Residual.Magnitude < 250 and Residual or Vector3.zero
    Sample.Forces, Sample.Time, Sample.Velocity = Active, Now, Velocity
    local Excluded = game:GetService("CollectionService"):GetTagged("Ball")
    if Dive.Enabled and Player.Character then Excluded[#Excluded + 1] = Player.Character end
    Cast.FilterDescendantsInstances = Excluded
    Cast.CollisionGroup = Ball.CollisionGroup
    local Points = Forecast(Ball, Sample, Cast)
    AutoDive(Ball, Points)
    if not Prediction.Enabled then HidePath() return end
    local Horizontal = Vector3.new(Velocity.X, 0, Velocity.Z)
    local Side = Horizontal.Magnitude > 0 and Total - Horizontal.Unit * Total:Dot(Horizontal.Unit) or Total
    local Curved = Shot.curve or Vector3.new(Side.X, 0, Side.Z).Magnitude / Ball.AssemblyMass > 2
    local Kind = Sample.Knuckle and "Knuckle (estimate)" or Shot.spinType == "backspin" and "Backspin"
        or Shot.risingShot and "Rising shot" or Shot.trivelaShot and "Trivela" or Shot.finesseShot and "Finesse"
        or Curved and "Curve" or "Straight"
    local Color = Sample.Knuckle and Color3.fromRGB(255, 181, 65) or Curved and Color3.fromRGB(192, 125, 255)
        or Color3.fromRGB(80, 220, 255)
    for i, v in pairs(Lines) do
        local From, To = Points[i], Points[i + 1]
        v.Visible = From ~= nil and To ~= nil and (To - From).Magnitude > 0.001
        if v.Visible then
            v.CFrame = workspace.Terrain.CFrame:ToObjectSpace(CFrame.lookAt(From, To))
            v.Length, v.Color3 = (To - From).Magnitude, Color
        end
    end
    Marker.Visible = #Points > 1
    Info.Enabled = Marker.Visible and Prediction.Names
    Marker.CFrame, Marker.Color3 = workspace.Terrain.CFrame:ToObjectSpace(CFrame.new(Points[#Points])), Color
    Info.StudsOffsetWorldSpace = Points[#Points] - workspace.Terrain.Position + Vector3.new(0, 1, 0)
    Label.Text, Label.TextColor3 = Kind, Color
end)

local ShotImpact = function(Ball, Args, Velocity, Goal, Depth, Sign)
    local Launch = Velocity
    local Speed = Vector3.new(Velocity.X, 0, Velocity.Z).Magnitude
    local Direction = Args.curveType or "none"
    local Angle = Args.lookAngleAtRelease or RootScript:GetAttribute("CurrentLookAngle_0_60")
    local Knuckle = Args.spinType == "knuckle" and (not Angle or Angle > 10)
    local Lateral = Direction == "left" or Direction == "right"
    local Flat = Velocity.Y <= 2 and not Args.simpleFinesseCurve
    local Duration = Speed / (Flat and 220 or Args.risingShot and 40 or 120)
    local Strength = (Args.finesseShot and 1.8 or Args.trivelaShot and 1.7 or 1) * (Args.curveMultiplier or 1)
    local Density = (Ball.CustomPhysicalProperties or Ball.CurrentPhysicalProperties).Density
    local Gravity = Ball:FindFirstChild("gravity")
    local Base = Vector3.new(0, -workspace.Gravity, 0) + (Gravity and Force(Gravity) or Vector3.zero) / Ball.AssemblyMass
    local Position, Step = Ball.Position, 1 / 120
    local Previous = (Goal.CFrame:PointToObjectSpace(Position).Z - Depth) * Sign
    for i = 1, 180 do
        local Time, Acceleration = (i - 0.5) * Step, Base
        if Args.curve and not (Flat and Time >= Duration * 3) then
            local Side = Direction == "left" and 1 or Direction == "right" and -1 or 0
            if Knuckle then
                Side = Direction == "right" and -1 or 1
                if Time >= Duration / 1.3 then Side = -Side end
            end
            local Spin = Vector3.new(Launch.Z, 0, -Launch.X).Unit * -16
                + Vector3.new(0, Side * 50 * Strength / (Knuckle and 15 or 1), 0)
            if Direction == "none" and not Knuckle then Spin *= 2.15 end
            local Curve = Spin:Cross(Args.simpleFinesseCurve and Vector3.new(Launch.X, 7, Launch.Z) or Launch)
            Curve = Curve.Magnitude > 0 and Curve.Unit * (500 * Density * Strength) or Vector3.zero
            Curve *= Knuckle and Vector3.new(1.5, 0.075, 1.5) or Vector3.new(0.5, 0, 0.5)
            if Args.risingShot then
                Curve *= Vector3.new(0.375, 1, 0.375)
                Curve += Vector3.new(0, -Base.Y * Ball.AssemblyMass * ((Args.power - 60) / 40), 0)
            end
            local Age = Time
            local Length, EndY = Duration * (Lateral and 1 or 1.15), Lateral and -0.5 or -0.25
            if Knuckle and Time >= Duration / 1.3 then Age, Length, EndY = Time - Duration / 1.3, Duration - Duration / 1.3, -0.5 end
            Curve *= Vector3.new(1, 1, 1):Lerp(Vector3.new(0.25, EndY, 0.25), math.clamp(Age / Length, 0, 1))
            Acceleration += Curve / Ball.AssemblyMass
        end
        local NextVelocity = Time < 0.3 and Launch or Velocity + Acceleration * Step
        local Next = Position + (Velocity + NextVelocity) * (Step / 2)
        local Current = (Goal.CFrame:PointToObjectSpace(Next).Z - Depth) * Sign
        if Current <= 0 then
            local Alpha = Previous / (Previous - Current)
            return Position:Lerp(Next, Alpha), (i - 1 + Alpha) * Step
        end
        Position, Velocity, Previous = Next, NextVelocity, Current
    end
end
local Shot = { React = React }
Shot.Func = function(Self, Args)
    local Ball, Team = Args.ball, Player.Team and Player.Team.Name or ""
    local Folder = workspace:FindFirstChild("game")
    Folder = Folder and Folder:FindFirstChild("points")
    Folder = Folder and Folder:FindFirstChild("goals")
    local Goal = Folder and Folder:FindFirstChild(Team:find("Home") and "away" or Team:find("Away") and "home" or "")
    if Options.Miss and Args.updateStats == "Shots" and Goal and Ball and Ball.Parent and not Ball.Anchored
        and typeof(Args.vector) == "Vector3" and (Goal.Position - Ball.Position).Magnitude <= 110 then
        local Velocity, Offset = Args.vector, Goal.Position - Ball.Position
        local Speed = Vector3.new(Velocity.X, 0, Velocity.Z).Magnitude
        local Radius = math.min(Ball.Size.X, Ball.Size.Y, Ball.Size.Z) / 2
        local Width, Height = Goal.Size.X / 2 - Radius - 1, Goal.Size.Y / 2 - Radius - 0.75
        local Local = Goal.CFrame:PointToObjectSpace(Ball.Position)
        local Sign = math.sign(Local.Z)
        local Depth = Sign * (Goal.Size.Z / 2 - Radius - 0.2)
        if Speed > 5 and Width > 0 and Height > 0 and math.abs(Local.Z) > Goal.Size.Z / 2 + Radius
            and Velocity:Dot(Offset) > Velocity.Magnitude * Offset.Magnitude * 0.5 then
            local Hit = ShotImpact(Ball, Args, Velocity, Goal, Depth, Sign)
            local Point = Hit and Goal.CFrame:PointToObjectSpace(Hit) or Vector3.zero
            if not Hit or math.abs(Point.X) > Width or math.abs(Point.Y) > Height then
                local Target = Goal.CFrame:PointToWorldSpace(Vector3.new(math.clamp(Point.X, -Width, Width), math.clamp(Point.Y, -Height, Height), Depth))
                local Aim = Velocity
                for i = 1, 5 do
                    local Impact, Time = ShotImpact(Ball, Args, Aim, Goal, Depth, Sign)
                    if not Impact then break end
                    if (Target - Impact).Magnitude < 0.1 then break end
                    Aim += (Target - Impact) / Time
                    local Flat = Vector3.new(Aim.X, 0, Aim.Z)
                    if Flat.Magnitude < 1 then break end
                    Aim = Flat.Unit * math.min(Speed, 140) + Vector3.new(0, math.clamp(Aim.Y, -80, 80), 0)
                end
                local Impact = ShotImpact(Ball, Args, Aim, Goal, Depth, Sign)
                local Safe = Impact and Goal.CFrame:PointToObjectSpace(Impact)
                if Safe and math.abs(Safe.X) <= Width + 0.15 and math.abs(Safe.Y) <= Height + 0.15 then Args.vector = Aim end
            end
        end
    end
    return React(Self, Args)
end

local LastReactTime, ReactBusy = 0, false

Env.Step = game:GetService("RunService").Stepped:Connect(function()
    Env.Sphere.Visible = Reach.Enabled and Reach.Visualizer
    Env.Sphere.Radius = Reach.Value / 2
    Env.Sphere.Transparency = Reach.Transparency
    local Character = Player.Character
    local Part = Character and Character:FindFirstChild("HumanoidRootPart")
    local Nearest, Distance = nil, math.huge
    local Owned, OwnedDistance = nil, math.huge
    local Tracked, TrackDistance = nil, math.huge
    local Threat, ThreatTime = nil, math.huge
    if not Part then
        Env.Sphere.Visible = false
        if Data.VisualizerBall and not Data.VisualizerFailed then
            Data.VisualizerFailed = not pcall(function() Env.Sphere.Adornee = nil end)
            Data.VisualizerBall = nil
        end
        PredictionBall = nil
        return
    end
    Params.FilterDescendantsInstances = { Character }
    for i, v in pairs(game:GetService("CollectionService"):GetTagged("Ball")) do
        if not v:IsA("BasePart") or not v:IsDescendantOf(workspace) then continue end
        local Range = (v.Position - Part.Position).Magnitude
        if Range < Distance then Nearest, Distance = v, Range end
        if not v.Anchored and not v:GetAttribute("catch") and Range < TrackDistance then
            Tracked, TrackDistance = v, Range
        end
        if Dive.Enabled and not v.Anchored and not v:GetAttribute("catch") then
            local Depth = (v.Position - Part.Position):Dot(Part.CFrame.LookVector)
            local Closing = -(v.AssemblyLinearVelocity - Part.AssemblyLinearVelocity):Dot(Part.CFrame.LookVector)
            if Depth > 0 and Closing > 8 and Depth / Closing < ThreatTime then
                Threat, ThreatTime = v, Depth / Closing
            end
        end
        if Options.BreakReact and Range <= 5.5 and Range < OwnedDistance
            and not v.Anchored and not v:GetAttribute("catch")
            and v:GetAttribute("networkOwner") == Player.UserId then
            Owned, OwnedDistance = v, Range
        end
        if not Reach.Enabled or Reach.Comp and v:GetAttribute("networkOwner") == Player.UserId then continue end
        for i, p in pairs(workspace:GetPartBoundsInRadius(v.Position, Reach.Value / 2, Params)) do
            if not p.Parent or p.Name:find("Collide") then continue end
            firetouchinterest(p, v, 0)
            firetouchinterest(p, v, 1)
        end
    end
    if not Data.VisualizerFailed and Data.VisualizerBall ~= Nearest then
        Data.VisualizerFailed = not pcall(function() Env.Sphere.Adornee = Nearest end)
        Data.VisualizerBall = Nearest
    end
    Env.Sphere.Visible = not Data.VisualizerFailed and Reach.Enabled and Reach.Visualizer and Nearest ~= nil
        and (not Reach.Comp or Nearest:GetAttribute("networkOwner") ~= Player.UserId)
    PredictionBall = Dive.Enabled and Threat or Tracked
    AutoCatch(Tracked)

    if Options.BreakReact and Network and Owned and not ReactBusy and os.clock() - LastReactTime >= 0.75 then
        local Head = Character:FindFirstChild("Head")
        if not Head then return end
        LastReactTime, ReactBusy = os.clock(), true
        task.spawn(function()
            if Options.BreakReact and Owned.Parent and Head.Parent and Player.Character == Character
                and Owned:GetAttribute("networkOwner") == Player.UserId then
                Fetch("networkOwner", {
                    checkOffside = false,
                    specialTool = true,
                    exemptShielding = true,
                    block = true,
                    ball = Owned,
                    limb = Head,
                    cframe = Owned.CFrame,
                    direction = "none",
                    sideLerp = 1,
                    vector = Owned.AssemblyLinearVelocity,
                    time = workspace:GetServerTimeNow(),
                })
            end
            ReactBusy = false
        end)
    end
end)
